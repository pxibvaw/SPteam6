# 그리고 가상환경을 만들 때 시스템 패키지를 보이게 해야 함:
#   python3 -m venv --system-site-packages .venv
# (gpiozero도 Pi OS에 기본 설치되어 있어서 같은 방법으로 사용)

-r requirements.txt
